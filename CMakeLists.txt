cmake_minimum_required(VERSION 3.20)
project(OmniOCR VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(OMNIOCR_WITH_ONNX "Build ONNX Runtime backend" OFF)
option(OMNIOCR_WITH_ACL "Build Ascend ACL backend" OFF)
option(OMNIOCR_BUILD_TESTS "Build tests" ON)
option(OMNIOCR_WITH_SERVER "Build REST API server (libmicrohttpd)" OFF)
set(ONNXRUNTIME_ROOT "" CACHE PATH "ONNX Runtime SDK root (include/, lib/)")
set(ASCEND_HOME "$ENV{ASCEND_HOME_PATH}" CACHE PATH "CANN toolkit root")
set(STB_INCLUDE_DIR "" CACHE PATH "Directory containing stb_image.h and stb_image_write.h")
include(FetchContent)
find_package(Threads REQUIRED)
find_package(CURL REQUIRED)
find_package(TIFF REQUIRED)
find_package(nlohmann_json 3.11 QUIET)
if(NOT nlohmann_json_FOUND)
    find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp)
    if(NLOHMANN_JSON_INCLUDE_DIR)
        add_library(omniocr_json INTERFACE)
        target_include_directories(omniocr_json INTERFACE "${NLOHMANN_JSON_INCLUDE_DIR}")
        add_library(nlohmann_json::nlohmann_json ALIAS omniocr_json)
    else()
        FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3 GIT_SHALLOW TRUE)
        FetchContent_MakeAvailable(json)
    endif()
endif()
if(NOT STB_INCLUDE_DIR)
    FetchContent_Declare(stb GIT_REPOSITORY https://github.com/nothings/stb.git GIT_TAG 2c980bb59875b0d32144a71867fbdebb2f77cd20)
    FetchContent_MakeAvailable(stb)
    set(STB_INCLUDE_DIR "${stb_SOURCE_DIR}")
endif()
add_library(omniocr_core
    src/config.cpp src/config_v2.cpp src/image.cpp src/model_pool.cpp src/layout.cpp src/plugins.cpp src/plugin_loader.cpp src/doclayout_v3.cpp src/geometry.cpp src/document.cpp
    src/csv.cpp src/tiff.cpp src/pipeline.cpp src/batch.cpp src/output.cpp src/table.cpp src/backends/http.cpp src/backends/models.cpp
    src/backends/tensor.cpp src/backends/onnx.cpp src/backends/acl.cpp)
target_include_directories(omniocr_core PUBLIC include PRIVATE "${STB_INCLUDE_DIR}")
target_link_libraries(omniocr_core PUBLIC nlohmann_json::nlohmann_json Threads::Threads PRIVATE CURL::libcurl ${CMAKE_DL_LIBS} TIFF::TIFF)
target_compile_options(omniocr_core PRIVATE -Wall -Wextra -Wpedantic)
# stb is external C code compiled as C++; keep its warning policy isolated.
set_source_files_properties(src/image.cpp PROPERTIES COMPILE_OPTIONS "-Wno-missing-field-initializers")
if(OMNIOCR_WITH_ONNX)
    find_path(ORT_INCLUDE_DIR onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include" REQUIRED)
    find_library(ORT_LIBRARY onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib" REQUIRED)
    target_include_directories(omniocr_core PRIVATE "${ORT_INCLUDE_DIR}")
    target_link_libraries(omniocr_core PRIVATE "${ORT_LIBRARY}")
    target_compile_definitions(omniocr_core PRIVATE OMNIOCR_WITH_ONNX)
endif()
if(OMNIOCR_WITH_ACL)
    find_path(ACL_INCLUDE_DIR acl/acl.h HINTS "${ASCEND_HOME}/include" REQUIRED)
    find_library(ACL_LIBRARY ascendcl HINTS "${ASCEND_HOME}/lib64" REQUIRED)
    target_include_directories(omniocr_core PRIVATE "${ACL_INCLUDE_DIR}")
    target_link_libraries(omniocr_core PRIVATE "${ACL_LIBRARY}")
    target_compile_definitions(omniocr_core PRIVATE OMNIOCR_WITH_ACL)
endif()
add_executable(omniocr src/main.cpp)
target_link_libraries(omniocr PRIVATE omniocr_core)
if(OMNIOCR_WITH_SERVER)
    find_path(MICROHTTPD_INCLUDE_DIR microhttpd.h REQUIRED)
    find_library(MICROHTTPD_LIBRARY NAMES microhttpd REQUIRED)
    add_executable(omniocr-server src/server.cpp)
    target_include_directories(omniocr-server PRIVATE "${MICROHTTPD_INCLUDE_DIR}")
    target_link_libraries(omniocr-server PRIVATE omniocr_core "${MICROHTTPD_LIBRARY}")
    target_compile_options(omniocr-server PRIVATE -Wall -Wextra -Wpedantic)
    install(TARGETS omniocr-server)
endif()
if(OMNIOCR_BUILD_TESTS)
    enable_testing()
    add_executable(omniocr_tests tests/core_tests.cpp)
    target_link_libraries(omniocr_tests PRIVATE omniocr_core)
    add_test(NAME core COMMAND omniocr_tests)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_FOUND)
        add_test(NAME http_integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/http_integration.py" $<TARGET_FILE:omniocr>)
        add_test(NAME fallback_http_integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/fallback_http_integration.py" $<TARGET_FILE:omniocr>)
    endif()
endif()
install(TARGETS omniocr omniocr_core)
install(DIRECTORY include/ DESTINATION include)
